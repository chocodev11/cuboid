// A screen that sets its own look.
fn show(ui: &mut egui::Ui) {
    let font = egui::FontId::proportional(13.5);
    ui.label(egui::RichText::new("x").size(12.0));
    let tint = egui::Color32::from_rgb(10, 20, 30);
    let white = egui::Color32::WHITE;
    ui.add_space(8.0);
    let frame = egui::Frame::new().inner_margin(egui::Margin::same(6));
    let round = egui::CornerRadius::same(3);
}
